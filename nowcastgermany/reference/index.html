    The German hospitalisation nowcasting hub as a case study.

    Import from the modules themselves, for example from nowcastgermany.evaluation import load_germany_config.

    nowcastgermany.data ¶

    The German COVID-19 hospitalisation nowcasting hub data as forecast tasks.

    The German hub (Wolffram et al. 2023) collected nowcasts of the 7-day hospitalisation incidence by Meldedatum, the date a case was reported positive, for Germany, its 16 federal states and six age groups. Its reporting triangle gives each Meldedatum's count as first reported, the change reported on each of the next 80 days, and the changes after that in one column. hub_triangle turns it into one row per Meldedatum, place and report date with the count reported by then, the layout of epinowcast's germany_covid19_hosp, which read_triangle reads.

    nowcast_tasks makes the tasks of one nowcast date in the ForecastTask form, so every registered model runs on them unchanged.

    • Each report date up to the nowcast date is a release with that date as its cut-off, and the Meldedatum is the onset date.
    • Germany is the country. The states are provinces in one task and the age groups provinces in another, each summing to Germany (place_code).
    • The hub's targets are the 7-day sums ending on every day from the nowcast date to 28 days before it, and the days after it to forecast (HORIZONS). A nowcast date gives one pair of tasks whose target dates are all of them, so every sum comes from one forecast with one context end. Dates are real: the nowcast date is the cut-off, whatever its weekday, and every model forecasts every target date.

    hub_forecasts moves the forecasts back and names the places as the hub does, and hub_target gives the 7-day sums as each report date gives them, the truth they are scored on.

    def place_code(location: str, age_group: str = ALL_AGES) -> str:
    """Return the location code a hub place has in the tasks.

    Parameters
    ----------
    location : str
        The hub's location, `DE` or a state such as `DE-BY`.

    def hub_changes(wide: pd.DataFrame) -> pd.DataFrame:
    """Return the hub's wide triangle as one change per reported delay.

    The changes after 80 days are at 81 days, as epinowcast's
    `germany_covid19_hosp` puts them. A missing change is a report date
    after the file was made, so it has no row.

    Parameters
    ----------
    wide : pandas.DataFrame
        The hub's `COVID-19_hospitalizations.csv`: columns `date`,
        `location`, `age_group`, `value_0d` to `value_80d` and
        `value_>80d`.

    Returns
    -------
    pandas.DataFrame
        `date`, `location`, `age_group`, `delay` (days) and `change`.
    """
    changes = [c for c in wide.columns if c.startswith("value_")]
    long = wide.melt(
        id_vars=["date", "location", "age_group"],
        value_vars=changes,
        var_name="delay",
        value_name="change",
    ).dropna(subset=["change"])
    long["delay"] = long["delay"].map({c: i for i, c in enumerate(changes)})
    return long

    def hub_triangle(path: Path | str) -> pd.DataFrame:
    """Read the hub's reporting triangle as counts by report date.

    The changes, cumulated, of `hub_changes`.

    Parameters
    ----------
    path : pathlib.Path or str
        The hub's `COVID-19_hospitalizations.csv`.

    Returns
    -------
    pandas.DataFrame
        As `read_triangle`.
    """
    long = hub_changes(pd.read_csv(path)).sort_values(
        ["location", "age_group", "date", "delay"]
    )
    long["confirm"] = (
        long.groupby(["location", "age_group", "date"])["change"]
        .cumsum()
        .astype("int64")
    )
    report = pd.to_datetime(long["date"]) + pd.to_timedelta(
        long["delay"], unit="D"
    )
    out = pd.DataFrame(
        {
            "reference_date": long["date"],
            "location": long["location"],
            "age_group": long["age_group"],
            "confirm": long["confirm"],
            "report_date": report.dt.date.astype(str),
        }
    )
    return out.reset_index(drop=True)

    def nowcast_tasks(
    triangle: pd.DataFrame,
    nowcast_date: str,
    n_samples: int,
    seed: int,
    horizons: Sequence[int] = HORIZONS,
) -> list[ForecastTask]:
    """Return the states and the age groups task of a nowcast date.

    Parameters
    ----------
    triangle : pandas.DataFrame
        Counts by Meldedatum and report date (`read_triangle`).
    nowcast_date : str
        The nowcast date, ISO format. Report dates after it are left out.
    n_samples : int
        Samples per forecast.
    seed : int
        Seed of the tasks' random numbers.
    horizons : sequence of int
        Days from the nowcast date to the end of each 7-day sum, the
        hub's (`HORIZONS`) by default. Days after it are forecasts.

    Returns
    -------
    list of ForecastTask
        The task of the states in the triangle, then that of the age
        groups, each with Germany as the country. The reference date and
        cut-off are the nowcast date, and the target dates are every
        horizon's day, so one forecast gives every 7-day sum.
    """
    # The hub has age groups for every state, but only Germany's are
    # nowcast.
    rows = triangle.loc[
        (triangle["report_date"] <= nowcast_date)
        & (
            (triangle["age_group"] == ALL_AGES)
            | (triangle["location"] == GERMANY)
        )
    ]
    rows = _carried(rows)
    rows = rows.loc[rows["confirm"] != 0]
    ref = as_date(nowcast_date)
    vintage = rows["report_date"]
    daily = pd.DataFrame(
        {
            "vintage": vintage,
            "location": _place_codes(rows["location"], rows["age_group"]),
            "onset_date": rows["reference_date"],
            "observed": rows["confirm"],
            "imputed": 0,
            "cutoff": vintage,
        }
    )[_DAILY]
    weekly = _weekly(daily)
    present = set(daily["location"].unique())
    groups = (
        tuple(place_code(s) for s in STATES if place_code(s) in present),
        tuple(
            place_code(GERMANY, a)
            for a in AGE_GROUPS
            if place_code(GERMANY, a) in present
        ),
    )
    days = tuple(
        (ref + dt.timedelta(days=h)).isoformat() for h in sorted(horizons)
    )
    tasks = []
    for places in groups:
        keep = {NATIONAL, *places}
        own = daily.loc[daily["location"].isin(keep)].reset_index(drop=True)
        history = weekly.loc[weekly["location"].isin(keep)]
        tasks.append(
            ForecastTask(
                vintage=ref.isoformat(),
                reference_date=ref,
                kind="real",
                cutoff=ref.isoformat(),
                target=history.loc[
                    history["vintage"] == ref.isoformat()
                ].reset_index(drop=True),
                delays=None,
                locations=places,
                target_end_dates=days,
                n_samples=n_samples,
                seed=seed,
                target_history=history.reset_index(drop=True),
                daily_history=own,
            )
        )
    return tasks

    def hub_target(triangle: pd.DataFrame, ends: Sequence[str]) -> pd.DataFrame:
    """Return 7-day sums as each report date gives them, as a target table.

    Parameters
    ----------
    triangle : pandas.DataFrame
        Counts by Meldedatum and report date (`read_triangle`).
    ends : sequence of str
        Last days of the 7-day sums.

    Returns
    -------
    pandas.DataFrame
        One row per place in `PLACE_NAMES`, end and report date from the
        end on: `kind` (`real`), `vintage` and `cutoff` (the report date),
        `location` (the place), `target_end_date` (the end) and `total`.
    """
    places = triangle["location"].where(
        triangle["age_group"] == ALL_AGES, triangle["age_group"]
    )
    keep = places.isin(set(PLACE_NAMES)) & (
        (triangle["age_group"] == ALL_AGES) | (triangle["location"] == GERMANY)
    )
    first = (as_date(min(ends)) - dt.timedelta(days=6)).isoformat()
    rows = triangle.loc[keep & (triangle["reference_date"] >= first)]
    reports = sorted(set(rows["report_date"]) | set(ends))
    days = pd.date_range(first, max(ends), freq="D").date.astype(str)
    frames = []
    for place, own in rows.groupby(places.loc[rows.index]):
        # Count of each Meldedatum (row) by each report date (column).
        counts = (
            own.pivot_table(
                index="reference_date",
                columns="report_date",
                values="confirm",
                aggfunc="last",
            )
            .reindex(index=days, columns=reports)
            .ffill(axis=1)
            .fillna(0.0)
        )
        sums = counts.rolling(7).sum()
        for end in ends:
            total = sums.loc[end]
            total = total.loc[total.index >= end]
            frames.append(
                pd.DataFrame(
                    {
                        "kind": "real",
                        "vintage": total.index,
                        "cutoff": total.index,
                        "location": place,
                        "target_end_date": end,
                        "total": total.to_numpy(),
                    }
                )
            )
    return pd.concat(frames, ignore_index=True)

    Score our models and the hub teams on the German nowcast benchmark.

    The benchmark is that of Wolffram et al.     def germany_forecasts(
    models: Sequence[str],
    triangle: pd.DataFrame,
    dates: Sequence[str],
    n_samples: int,
    seed: int,
    tasks_dir: Path | None = None,
    horizons: Sequence[int] = HORIZONS,
    max_delay: int | None = None,
    timings: list[dict[str, Any]] | None = None,
    targets: Sequence[str] = ("7day",),
) -> pd.DataFrame:
    """Nowcast each date with each model, as quantiles at the hub's levels.

    Parameters
    ----------
    models : sequence of str
        Registered model names.
    triangle : pandas.DataFrame
        Counts by Meldedatum and report date
        (`nowcastgermany.data.read_triangle`).
    dates : sequence of str
        Nowcast dates.
    n_samples, seed : int
        Samples per forecast where the quantiles come from samples, and
        their seed, the same for every model.
    tasks_dir : pathlib.Path, optional
        A store of each date's tasks (`<date>.pkl`): read where it holds
        one, else built and written to it. The tasks are the same for
        every checkpoint, so they are built once (`checkpoint.german_tasks_dir`
        keys the store on what they are built from).
    horizons : sequence of int
        Days from each nowcast date to the end of each 7-day sum, the
        hub's 0 to -28 (`HORIZONS`) by default; days after it are
        forecasts (`nowcastgermany.data.nowcast_tasks`).
    max_delay : int, optional
        The truth's delay (`GermanyConfig.truth_delay`): the models see
        only the counts reported within it (`nowcastgermany.data.within_delay`),
        so they forecast the count the truth holds, not the eventual one.
        A `tasks_dir` must then be keyed on it.
    timings : list, optional
        Given, a row is appended for each model and date: `model`,
        `origin`, `stage` (`forecast`) and `seconds`; and one for each
        Chronos-2 model's batched pass over every date (`stage`
        `prefetch`, `origin` empty).
    targets : sequence of str
        Views of `TARGETS` to forecast. `daily` rows are each day's count,
        from the same forecast as the 7-day sums, as a model's own
        quantiles of its daily count rows
        (`FoundationModel.daily_quantiles`); a model without them gives
        none.

    Returns
    -------
    pandas.DataFrame
        One row per quantile: `model`, `target`, `origin` (the nowcast date),
        `location` (the place), `target_end_date`, `horizon` (days),
        `quantile` and `value`, at `FORECAST_LEVELS`
        (`nowcasteval.marginals.forecast_quantiles`).
    """
    if max_delay is not None:
        triangle = within_delay(triangle, max_delay)
    built = [get_model(m) for m in models]
    # Chronos-2 is predicted for every date's tasks in a few batched passes,
    # so it needs them all at once. Each task holds its own copy of the
    # release history, so other models build one date's tasks at a time.
    tasks: dict[str, list[ForecastTask]] = {}
    if any(isinstance(m, FoundationModel) for m in built):
        tasks = {
            d: _tasks(triangle, d, n_samples, seed, tasks_dir, horizons)
            for d in dates
        }
        every = [t for ts in tasks.values() for t in ts]
        built = [
            _timed_prefetch(name, m, every, timings)
            if isinstance(m, FoundationModel)
            else m
            for name, m in zip(models, built, strict=True)
        ]
        # The loop pops each date's tasks, which frees them only once this
        # list no longer holds them.
        del every
    keys = [
        "model", "target", "origin", "location", "target_end_date", "horizon"
    ]  # fmt: skip
    daily = "daily" in targets
    frames = []
    for date in dates:
        mine = tasks.pop(date, None)
        if mine is None:
            mine = _tasks(triangle, date, n_samples, seed, tasks_dir, horizons)
        spent = dict.fromkeys(models, 0.0)
        for task in mine:
            for name, model in zip(models, built, strict=True):
                start = time.perf_counter()
                at = forecast_quantiles(
                    model, task, FORECAST_LEVELS, n_samples, seed
                )
                spent[name] += time.perf_counter() - start
                if at.empty:
                    continue
                # `hub_forecasts` moves rows to the hub's dates and places,
                # carrying each row's level where it carries a sample.
                moved = hub_forecasts(
                    at.rename(columns={"quantile": "sample"}), task, date
                )
                frames.append(
                    moved.rename(
                        columns={
                            "forecast_date": "origin",
                            "sample": "quantile",
                        }
                    ).assign(target="7day")
                )
        if daily:
            frames += _daily_rows(models, built, mine, date)
        if timings is not None:
            timings += [
                {"model": m, "origin": date, "stage": "forecast", "seconds": t}
                for m, t in spent.items()
            ]
    if not frames:
        return pd.DataFrame(columns=[*keys, "quantile", "value"])
    out = pd.concat(frames, ignore_index=True)
    return out.sort_values([*keys, "quantile"], kind="stable")[
        [*keys, "quantile", "value"]
    ].reset_index(drop=True)

    def relative_scores(
    scores: pd.DataFrame, by: Sequence[str], baseline: str

    """Return each model's pairwise relative scores and its own means.

    The relative scores are `nowcasteval.scoring.pairwise_skill`, so
    models that forecast different nowcasts (hub teams that missed
    dates) compare fairly. The means are over each model's own
    forecasts.

    Parameters
    ----------
    scores : pandas.DataFrame
        `score_germany` output, for our models and the teams.
    by : sequence of str
        Columns of a cell other than `model`.
    baseline : str
        The model the relative scores are scaled to, such as
        `GermanyConfig.baseline`.

    Returns
    -------
    pandas.DataFrame
        One row per cell and model: `n` (its forecasts), `rounds` (its
        nowcast dates), `wis`, `rel_wis`, `ae_median`, `rel_ae_median`,
        and the coverage of the 50% and 95% intervals.
    """
    by = list(by)
    match = ("target", "origin", "truth_vintage", "location", "target_end_date")
    rel = pairwise_skill(
        scores,
        baseline,
        by=by,
        metrics=("wis", "ae_median"),
        match_on=match,
        rounds="origin",
    )
    mean_cols = ["wis", "ae_median", "coverage_50", "coverage_95"]
    means = scores.groupby([*by, "model"])[mean_cols].mean().reset_index()
    rel = rel.merge(means, on=[*by, "model"]).rename(
        columns={
            "relative_wis": "rel_wis",
            "relative_ae_median": "rel_ae_median",
        }
    )
    cols = ["n", "rounds", "wis", "rel_wis", "ae_median", "rel_ae_median"]
    return rel[[*by, "model", *cols, "coverage_50", "coverage_95"]]

        complete_through=str(truth["target_end_date"].max()),
        data_label=f"reported {config.truth_delay} days on",
        asof=seen,

        "quantile",
    ]

    nowcastgermany.checkpoint ¶

    Score each published Chronos-2 checkpoint.

    Each checkpoint (release chronos-ft-<hash> or chronos-slots-<hash>) is a label, run beside reference (randomwalknowcast and zero-shot Chronos-2, ragged and complete). A label's units forecast and cache; table scores the caches into rows of evidence/checkpoint-scores.csv with rule = RULE:

    out = mean.reset_index().assign(study=study, rule=RULE, target=target)
    return out.reindex(columns=COLUMNS)

    # A date's rows are kept only if forecast from this task store: the
    # tasks' building code is in its name (`german_tasks_dir`). The
    # reference's do not depend on a checkpoint, so each model's are also
    # keyed on its own code and the German config (`model_key`), and a
    # change redoes that model's rows alone.
    tasks = tasks_dir.name
    want = {
        m: f"{tasks}-{model_key(m)}" if label == REFERENCE else tasks
        for m in models
    }
    have = written(root)
    groups: dict[tuple[str, ...], list[str]] = {}
    for d in dates:
        full = not _reaches(root / f"{d}.csv.gz", last, config.targets)
        need = tuple(
            m
            for m in models
            if full or have.get((d, m), have.get((d, EVERY))) != want[m]
        )
        if need:
            groups.setdefault(need, []).append(d)
    # Rows of a model the label no longer runs (a renamed or dropped
    # model) go from every date, also those nothing is redone on.
    gone = {d for (d, m) in have if m != EVERY and m not in models}
    for d in sorted(gone - {d for ds in groups.values() for d in ds}):
        out = root / f"{d}.csv.gz"
        if out.exists():
            old = pd.read_csv(out)
            _write(old.loc[old["model"].isin(models)], out)
    todo = [(need, d) for need, ds in groups.items() for d in ds]
    # Dates are forecast `score_dates()` at a time, Chronos-2 predicted for
    # all of them in a few batched passes, and written one file a date so
    # a stopped run resumes.
    step = score_dates()
    chunks = [
        (need, ds[i : i + step])
        for need, ds in groups.items()
        for i in range(0, len(ds), step)
    ]
    for i, (need, chunk) in enumerate(chunks):
        start = time.monotonic()
        timings: list[dict[str, Any]] = []
        fc = de.germany_forecasts(
            list(need),
            triangle,
            chunk,
            config.n_samples,
            config.seed,
            tasks_dir,
            max_delay=config.truth_delay,
            horizons=config.horizons,
            targets=config.targets,
            timings=timings,
        )
        # Seconds by model and date, for the `benchmarks` target.
        times = root / "timings.csv"
        pd.DataFrame(timings).assign(dates=len(chunk)).to_csv(
            times, mode="a", header=not times.exists(), index=False
        )
        fc = de.with_truth(fc, triangle, config.truth_delay)
        for date in chunk:
            mine = fc.loc[fc["origin"] == date].reset_index(drop=True)
            out = root / f"{date}.csv.gz"
            if len(need) < len(models) and out.exists():
                # The other models' rows of the date are kept.
                old = pd.read_csv(out)
                kept = old["model"].isin(models) & ~old["model"].isin(need)
                mine = pd.concat([old.loc[kept], mine], ignore_index=True)
            _write(mine, out)
            _record(root, date, {m: want[m] for m in need})
        _free()
        took = time.monotonic() - start
        done = len(dates) - len(todo) + sum(len(c) for _, c in chunks[: i + 1])
        print(
            f"germany {label} {len(chunk)} dates to {chunk[-1]} "
            f"({done}/{len(dates)}): {took:.0f} s",
            flush=True,
        )
    print(f"germany {label} done", flush=True)

        for task in map(
            hub_weeks, nowcast_tasks(seen, date, cfg.n_samples, cfg.seed)
        ):
            level = (
                "ages" if set(task.locations) & set(AGE_GROUPS) else "states"
            )
            for name in MODELS:
                drawn, fit = variant_forecasts(name, task)
                if fit is not None:
                    fits.append({**fit, "origin": date, "level": level})
                if not drawn:
                    continue
                moved = {
                    label: hub_forecasts(
                        s.assign(model=name, as_of=np.nan), task, date
                    )
                    for label, s in drawn.items()
                }
                ends = sorted(
                    set(next(iter(moved.values()))["target_end_date"])
                )
                truth = rolling_truth(
                    hub_target(triangle, ends), cfg.truth_delay
                )
                for label, s in moved.items():
                    frames.append(
                        variant_scores(
                            s,
                            truth,
                            study="germany",
                            model=name,
                            variant=label,
                            origin=date,
                            levels={
                                **levels,
                                **{
                                    p: "states"
                                    for p in set(s["location"])
                                    if p not in levels and p != "DE"
                                },
                            },
                        )
                    )
    rho = pd.DataFrame(fits, columns=list(RHO_COLUMNS))
    if not frames:
        return pd.DataFrame(columns=list(SCORE_COLUMNS)), rho
    out = pd.concat(frames, ignore_index=True)
    # Germany's provinces are its states.
    out["level"] = out["level"].replace({"provinces": "states"})
    return out, rho

    def cut_dates(first: str, truth_until: str) -> list[str]:
    """Return every nowcast date whose horizon ends by `truth_until`."""
    last = dt.date.fromisoformat(truth_until) - dt.timedelta(weeks=AHEAD_WEEKS)
    day, out = dt.date.fromisoformat(first), []
    while day <= last:
        out.append(day.isoformat())

    def german_counts(wide: pd.DataFrame, as_of: str) -> pd.DataFrame:
    """Return the German hub triangle's counts as reported by `as_of`.

    Each change of `value_0d` to `value_80d` is reported on its Meldedatum
    plus its delay; the change after 80 days is left out. States (all
    ages) and Germany's age groups are kept, named as the German tasks
    name them (`nowcastgermany.data.place_code`); Germany is their sum.

    Returns
    -------
    pandas.DataFrame
        `location`, `reference_date`, `report_date` and `count`, as
        `finetune.training_block` reads.
    """
    long = de.hub_changes(wide)
    long = long.loc[long["delay"] <= 80]
    states = (long["age_group"] == de.ALL_AGES) & (
        long["location"] != de.GERMANY
    )
    ages = (long["location"] == de.GERMANY) & (long["age_group"] != de.ALL_AGES)
    long = long.loc[states | ages]
    report = pd.to_datetime(long["date"]) + pd.to_timedelta(long["delay"], "D")
    out = pd.DataFrame(
        {
            "location": [
                de.place_code(loc, age)
                for loc, age in zip(
                    long["location"], long["age_group"], strict=True
                )
            ],
            "reference_date": long["date"].astype(str),
            "report_date": report.dt.date.astype(str),
            "count": long["change"].astype("int64"),
        }
    )
    out = out.loc[out["report_date"] <= as_of]
    return out.sort_values(
        ["location", "reference_date", "report_date"]
    ).reset_index(drop=True)

    def german_blocks(
    counts: pd.DataFrame,
    cuts: list[str],
    truth_until: str,
    first: int = 0,
) -> list[dict[str, Any]]:
    """Return the training blocks of the German data at nowcast dates.

    Each cut is laid out as the German tasks lay it out, on its real
    dates: every report date up to it is a release, and Germany is the
    country (`parent` exact). Each of `PLACES` gives a
    block by `finetune.training_block`, at a lag taken in turn from
    `LAGS` (from cut `first` on), padded as the simulated blocks are.

    Raises
    ------
    ValueError
        If a cut's horizon ends after `truth_until`.
    """
    places = sorted(counts["location"].unique())
    states = [p for p in places if "-A" not in p]
    ages = [p for p in places if "-A" in p]
    leaves = {"national": states, "states": states, "ages": ages}
    out = []
    for i, cut in enumerate(cuts, start=first):
        end = dt.date.fromisoformat(cut) + dt.timedelta(weeks=AHEAD_WEEKS)
        if end.isoformat() > truth_until:
            raise ValueError(
                f"cut {cut} needs truth to {end}, past the truth date "
                f"{truth_until}"
            )
        cutoffs = sorted(d for d in counts["report_date"].unique() if d <= cut)
        for j, which in enumerate(PLACES):
            view = (NATIONAL,)
            if which != "national":
                view = (NATIONAL, *leaves[which])
            own = counts.loc[counts["location"].isin(leaves[which])]
            block = training_block(
                own,
                SOURCE,
                view,
                cutoffs,
                max_lag=LAGS[(i + j) % len(LAGS)],
            )
            if block is None or block["start"] < 2:
                continue
            padded = ft._pad(
                block,
                ft.CONTEXT,
                ft.LOSS_HORIZON_DAYS,
                ft.DAYS_PER_STEP[SOURCE],
            )
            levels = ft.target_levels(view)
            padded["levels"] = tuple(levels[k] for k in block["target_places"])
            padded["places"] = view
            padded["target_places"] = list(block["target_places"])
            padded["parent"] = "exact"
            padded["mix"] = "germany"
            padded["source"] = SOURCE
            padded["cut"] = cut
            out.append(padded)
    return out

    def make_blocks(
    triangle: Path,
    out: Path,
    first_date: str,
    truth_delay: int,
    workers: int = 4,
) -> int:
    """Cut the German blocks of `window` and pickle them in cut order."""
    as_of, truth_until = window(first_date, truth_delay)
    counts = german_counts(pd.read_csv(triangle), as_of)
    cuts = cut_dates(FIRST_CUT, truth_until)
    jobs = [(counts, c, i, truth_until) for i, c in enumerate(cuts)]
    with ProcessPoolExecutor(workers) as pool:
        blocks = [b for part in pool.map(_cut, jobs) for b in part]
    with out.open("wb") as f:
        pickle.dump(blocks, f)
    print(
        f"{len(blocks)} blocks of {len(cuts)} cuts {cuts[0]} to {cuts[-1]}, "
        f"reports to {as_of}, truth to {truth_until}: {out}",
        flush=True,
    )
    return len(blocks)
