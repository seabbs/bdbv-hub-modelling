    nowcastutils ¶

    Shared pieces for nowcasting and forecasting.

    Import from the modules themselves, for example from nowcastutils.base import ForecastTask.

    nowcastutils.releases ¶

    Release cut-offs and the renewal model's report to publication lag.

    A release's data cut-off is its latest onset day over all places. The last day or two before extraction are thin, as new reports take time to reach the line list. latest_onset_days derives the cut-off from the data, whatever the label, and every tidy table carries it as a cutoff column, which release_cutoff reads. release_pairs tabulates, for each pair of consecutive delay files, which rows of the later file were already in the earlier one, the data the renewal models learn their report to publication lag from. cumulative_first_release finds a first release that is a cumulative dump rather than part of the reporting, and without_cumulative_first_release leaves it out of a task's releases.

    MAX_PAIR_AGE module-attribute ¶

    MAX_PAIR_AGE = 60

    Oldest report age at the earlier release of a pair kept by release_pairs, in days, the renewal models' maximum delay.

    def latest_onset_days(daily: pd.DataFrame) -> pd.Series:
    """Return each release's data cut-off, its latest onset day.

    The cut-off is the latest onset day over all places in the release's
    daily onsets, one date per release.

    Parameters
    ----------
    daily : pandas.DataFrame
        Daily onsets by release, with `vintage` and `onset_date`.

    Returns
    -------
    pandas.Series
        The cut-off (ISO) by release label, for every release in `daily`.
    """
    onset = pd.to_datetime(daily["onset_date"]).groupby(daily["vintage"]).max()
    out: pd.Series = onset.dt.strftime("%Y-%m-%d").rename("cutoff")
    return out

    def release_cutoffs(
    rows: pd.DataFrame, releases: Sequence[str]
) -> list[dt.date]:
    """Return each release's data cut-off, as `release_cutoff` does.

    One pass over `rows`, where `release_cutoff` compares every row for
    each release.

    Parameters
    ----------
    rows : pandas.DataFrame
        Rows with `vintage` and `cutoff` columns.
    releases : sequence of str
        The release labels.

    Returns
    -------
    list of datetime.date
        The cut-offs, in the order of `releases`.

    Raises
    ------
    KeyError
        If a release has no rows.
    """
    first = rows.drop_duplicates("vintage")
    at = {
        str(v): str(c)
        for v, c in zip(first["vintage"], first["cutoff"], strict=True)
    }
    return [as_date(at[str(r)]) for r in releases]

    def release_pairs(
    delays: pd.DataFrame,
    as_of: dt.date | str,
    max_age: int = MAX_PAIR_AGE,
) -> dict[str, np.ndarray] | None:
    r"""Rows of each delay file and whether the previous file had them.

    For each pair of consecutive delay files $k$ and $k + 1$ on or before
    `as_of`, every row cell (location, onset week, delay) of file $k + 1$
    is counted with how many of its rows file $k$ already had. A row with
    onset week ending $W$ and delay $d$ was reported on $W - u + d$ for
    an onset $u = 0..6$ days before $W$, so its report age at a release
    with cut-off $T$ is $T - W + u - d$. Cells that no onset day
    could have reported by $T_k$, or that are older than `max_age` at
    $T_k$ for every onset day, are left out.

    Parameters
    ----------
    delays : pandas.DataFrame
        Delay rows of several releases, with `vintage`, `cutoff`,
        `location`, `onset_week` and `delay`.
    as_of : datetime.date or str
        The forecast date. Later files are not used.
    max_age : int, optional
        Oldest report age kept, in days.

    Returns
    -------
    dict or None
        `before` and `after`, report ages at $T_k$ and $T_{k + 1}$ by
        onset day, `(n, 7)`; `kept` and `total` rows of each cell,
        `(n,)`; and `pair`, the one-hot pair of each cell, `(n, K)`.
        None if no pair of files is available.
    """
    day = as_date(as_of).isoformat()
    vintages = sorted(str(v) for v in delays["vintage"].unique() if v <= day)
    key = ["location", "onset_week", "delay"]
    parts = []
    for a, b in itertools.pairwise(vintages):
        seen = delays.loc[delays["vintage"] == a].groupby(key).size()
        rows = (
            delays.loc[delays["vintage"] == b]
            .groupby(key)
            .size()
            .reset_index(name="total")
        )
        had = seen.reindex(
            pd.MultiIndex.from_frame(rows[key]), fill_value=0
        ).to_numpy()
        # Report day of each row for an onset 0 to 6 days before the end
        # of its onset week, as days since the epoch.
        report = (
            pd.to_datetime(rows["onset_week"]).to_numpy().astype("datetime64[D]")
            .astype(int)[:, None]
            - np.arange(7)[None, :]
            + rows["delay"].to_numpy()[:, None]
        )  # fmt: skip

        def age(release: str, report: np.ndarray = report) -> np.ndarray:
            cut = np.datetime64(release_cutoff(delays, release), "D")
            return np.asarray(cut.astype(int) - report, dtype=int)

        before = age(a)
        keep = (before.max(axis=1) >= 0) & (before.min(axis=1) <= max_age)
        total = rows["total"].to_numpy()
        parts.append(
            {
                "before": before[keep],
                "after": age(b)[keep],
                "kept": np.minimum(had, total)[keep],
                "total": total[keep],
            }
        )
    if not parts:
        return None
    out = {k: np.concatenate([p[k] for p in parts]) for k in parts[0]}
    out["kept"] = out["kept"].astype(float)
    out["total"] = out["total"].astype(float)
    sizes = [len(p["total"]) for p in parts]
    out["pair"] = np.repeat(np.eye(len(parts)), sizes, axis=0)
    return out

    def cumulative_first_release(
    rows: pd.DataFrame | None, max_delay: int | None
) -> str | None:
    """Return the first release if it is a cumulative dump, else None.

    The first release is a cumulative dump when the next release's
    cut-off is more than $D$ (`max_delay`) days after its own. Every case
    it holds is then older than $D$ by the next release, so it shares no
    report within the delay window with any later release: it records
    the counts to date, not how they were reported. How far its onsets
    reach back is no test, as the first release of any data that start
    after the outbreak reaches back to its start, and so does the release
    after it.

    Parameters
    ----------
    rows : pandas.DataFrame or None
        Releases, with `vintage` and `cutoff`.
    max_delay : int or None
        $D$ in days. None finds no dump.

    Returns
    -------
    str or None
        The first release's label, or None if it is not a dump, there is
        no later release or no $D$.
    """
    if max_delay is None or rows is None or rows.empty:
        return None
    cut = rows.drop_duplicates("vintage")
    cut = cut.assign(cutoff=pd.to_datetime(cut["cutoff"])).sort_values("cutoff")
    if len(cut) < 2:
        return None
    gap = (cut["cutoff"].iloc[1] - cut["cutoff"].iloc[0]).days
    return str(cut["vintage"].iloc[0]) if gap > max_delay else None

    def without_cumulative_first_release(task: ForecastTask) -> ForecastTask:
    """Return the task without a cumulative first release in its releases.

    The first release of the task's releases (its daily releases, else
    its weekly ones) is left out of every release table
    (`HISTORY_FIELDS`) when it is a cumulative dump
    (`cumulative_first_release`, with the task's `max_delay`), so the
    history starts at the next release. Logged when it is.

    Parameters
    ----------
    task : ForecastTask
        The task.

    Returns
    -------
    ForecastTask
        The task, unchanged if its first release is not a dump.
    """
    daily = task.daily_history
    rows = daily if daily is not None and not daily.empty else None
    first = cumulative_first_release(
        task.target_history if rows is None else rows, task.max_delay
    )
    if first is None:
        return task
    log.info(
        "%s: release %s left out, a cumulative first release more than "
        "%d days before the next",
        task.vintage,
        first,
        task.max_delay,
    )
    tables: dict[str, Any] = {}
    for name in HISTORY_FIELDS:
        table = getattr(task, name)
        if table is not None and "vintage" in table.columns:
            kept = table.loc[table["vintage"].astype(str) != first]
            tables[name] = kept.reset_index(drop=True) if len(kept) else None
    return dataclasses.replace(task, **tables)

    nowcastutils.snapshots ¶

    Daily release snapshots as arrays on the onset clock.

    The hub's data/daily.csv holds one table of daily onsets for each release. A release $r$ has an as-of date $A_r$ and a cut-off $c_r$, its last report day (nowcastutils.releases.release_cutoff). Its cell for onset day $t$ is observed when $t \le c_r$. An observed day with no row counts 0. The age of day $t$ at release $r$ is $c_r - t$, the onset clock. Counts are observed plus imputed onsets, which sum to the hub's weekly target. Hub weeks run Sunday to Saturday and are labelled by the Saturday.

    def snapshot_arrays(
    daily: pd.DataFrame, vintage: str, locations: tuple[str, ...]
) -> SnapshotArrays:
    """Build the arrays of the releases on or before a vintage.

    Parameters
    ----------
    daily : pandas.DataFrame
        Daily onsets by release, with columns `vintage`, `location`,
        `onset_date`, `observed` and `imputed`.
    vintage : str
        The forecast vintage. Later releases are not used.
    locations : tuple of str
        Locations to include. Each location's ancestors
        (`nowcastutils.hierarchy.parent`) are added, so the root
        is always the first node. A node with no rows in `daily` is
        the sum of its children.

    Returns
    -------
    SnapshotArrays
        The arrays.

    Raises
    ------
    ValueError
        If no release is on or before the vintage.
    """
    rows = daily.loc[daily["vintage"].astype(str) <= vintage]
    if rows.empty:
        raise ValueError(f"No daily releases on or before {vintage}.")
    releases = tuple(sorted(str(v) for v in rows["vintage"].unique()))
    nodes: dict[str, None] = {}
    for loc in locations:
        chain: list[str] = []
        code: str | None = loc
        while code is not None and code not in nodes:
            chain.append(code)
            code = parent_of(code)
        for c in reversed(chain):
            nodes[c] = None
    nodes_t = tuple(sorted(nodes, key=lambda c: (depth(c), len(c), c)))
    first = as_date(str(rows["onset_date"].min()))
    cutoffs = release_cutoffs(rows, releases)
    days = pd.date_range(first, epiweek_end(max(cutoffs)), freq="D")
    index = {d: i for i, d in enumerate(days.date)}
    cut = np.array([index[c] for c in cutoffs], dtype=np.int64)
    observed = np.arange(len(days))[None, :] <= cut[:, None]
    counts = np.zeros((len(nodes_t), len(releases), len(days)))
    take = rows.loc[rows["location"].isin(nodes_t)]
    li = take["location"].map({c: i for i, c in enumerate(nodes_t)})
    ri = take["vintage"].astype(str).map({r: i for i, r in enumerate(releases)})
    ti = pd.to_datetime(take["onset_date"]).dt.date.map(index)
    values = (take["observed"] + take["imputed"]).to_numpy(dtype=float)
    np.add.at(
        counts,
        (
            li.to_numpy(dtype=int),
            ri.to_numpy(dtype=int),
            ti.to_numpy(dtype=int),
        ),
        values,
    )
    counts = counts * observed[None]
    ends = [epiweek_end(d) for d in days.date]
    week_ends = tuple(sorted({e.isoformat() for e in ends}))
    where = {w: i for i, w in enumerate(week_ends)}
    week = np.array([where[e.isoformat()] for e in ends], dtype=np.int64)
    pos = {c: i for i, c in enumerate(nodes_t)}
    parent = np.array(
        [
            pos[p] if (p := parent_of(c)) is not None and p in pos else -1
            for c in nodes_t
        ],
        dtype=np.int64,
    )
    # A node with no rows is the sum of its children, deepest first.
    given = set(take["location"])
    for i in reversed(range(len(nodes_t))):
        if nodes_t[i] not in given:
            counts[i] = counts[parent == i].sum(axis=0)
    return SnapshotArrays(
        nodes_t, releases, days, counts, observed, cut, week,
        week_ends, parent,
    )  # fmt: skip

        week_ends=week_ends, period=period,

    def reference_date_for_vintage(vintage: dt.date | str) -> dt.date:
    """Return the first reference date a data vintage can be submitted to.

    This is the earliest Saturday whose submission window closes on or
    after the vintage date.
    A vintage released on a Saturday serves the round ending a week later.
    One released on a Friday also serves the round ending eight days later,
    because its own round's window closed on the Tuesday.

    Parameters
    ----------
    vintage : datetime.date or str
        The `as_of` date of the data.

    Returns
    -------
    datetime.date
        The reference date, a Saturday.

    Examples
    --------
    >>> reference_date_for_vintage("2026-10-24")
    datetime.date(2026, 10, 31)
    >>> reference_date_for_vintage("2026-08-18")
    datetime.date(2026, 8, 22)
    """
    day = as_date(vintage)
    ref = epiweek_end(day)
    while ref + dt.timedelta(days=SUBMISSION_END_OFFSET) < day:
        ref += dt.timedelta(days=7)
    return ref

    nowcastutils.trajectories ¶

    Coherent sample trajectories from any model's quantile forecasts.

    A quantile forecast gives each series and step its own marginal and nothing about how they move together. couple draws trajectories whose cells each follow the cell's quantile function, piecewise linear between the levels and flat beyond them (so the end quantiles carry the tail mass), and joins the cells by a copula:

    • independent: every cell draws its own level;
    • comonotone: one level per sample and series across its steps, the quantile-only analogue of ensemble copula coupling;
    • gaussian: levels from a Gaussian copula with AR(1) correlation rho between steps and exchangeable correlation s between series, given in fit;
    • schaake: the Schaake shuffle, each sample's levels the ranks of one past window of values (fit["windows"]), observed (history_windows) or simulated.

    The parameters come from a fit; nothing is hard-coded. fit_coupling estimates them from normal scores of the model's own matured forecasts (pit_scores), and choose_coupling picks the coupling and how many of those origins to fit it from by the energy score (energy_score) of each candidate's draws on the matured earlier forecasts. This module depends on numpy, scipy and nowcastutils.dates only, so any model can use it.

    METHODS: Final = ('independent', 'comonotone', 'gaussian', 'schaake')

    How couple joins the cells.

    CANDIDATES module-attribute ¶

    CANDIDATES: Final = (('gaussian', 14), ('gaussian', 28), ('gaussian', 56), ('schaake', None), ('comonotone', None))

                f"got rho={rho}, s={s}"

    return _read_sorted(levels, values, u)

    quantiles: np.ndarray,
    levels: np.ndarray,
    observed: np.ndarray,
    rng: np.random.Generator,

    r"""Return normal scores of observed values in quantile forecasts.

    The score is $\Phi^{-1}$ of the randomised PIT,
    $F(y^-) + v\,(F(y) - F(y^-))$ with $v$ uniform, in the quantile
    function `read_quantiles` reads. So a value on a flat stretch (zero
    counts with every quantile zero, or the mass beyond the end levels)
    draws its level rather than taking a fixed one.

    Parameters
    ----------
    quantiles : numpy.ndarray
        Quantiles `[..., Q]` at `levels`.
    levels : numpy.ndarray
        Increasing quantile levels, `[Q]`.
    observed : numpy.ndarray
        Observed values, the shape of `quantiles` less its last axis; NaN
        where none is observed yet.
    rng : numpy.random.Generator
        Draws the uniforms.

    Returns
    -------
    numpy.ndarray
        Scores, the shape of `observed`, NaN where it is.
    """
    levels = np.asarray(levels, dtype=float)
    values = np.sort(np.asarray(quantiles, dtype=float), axis=-1)
    y = np.asarray(observed, dtype=float)
    u, v = _knots(levels, values)
    k = len(u)
    filled = np.where(np.isnan(y), 0.0, y)[..., None]

    def between(j: np.ndarray) -> np.ndarray:
        """Return the level at `filled` on the segment from knot j to j + 1."""
        j = np.clip(j, 0, k - 2)
        v0 = np.take_along_axis(v, j[..., None], axis=-1)[..., 0]
        v1 = np.take_along_axis(v, j[..., None] + 1, axis=-1)[..., 0]
        width = np.where(v1 > v0, v1 - v0, 1.0)
        t = np.clip((filled[..., 0] - v0) / width, 0.0, 1.0)
        return np.asarray(u[j] + (u[j + 1] - u[j]) * t)

    # F(y): the last knot at or below y; F(y-): the first at or above it.
    last = (v <= filled).sum(axis=-1) - 1
    upper = np.where(last >= k - 1, 1.0, np.where(last < 0, 0.0, between(last)))
    first = (v < filled).sum(axis=-1)
    lower = np.where(
        first >= k, 1.0, np.where(first <= 0, 0.0, between(first - 1))
    )
    pit = lower + rng.random(y.shape) * (upper - lower)
    tiny = np.finfo(float).eps
    z = ndtri(np.clip(pit, tiny, 1 - tiny))
    return np.where(np.isnan(y), np.nan, z)

    def history_windows(
    values: np.ndarray, steps: int, n_windows: int
) -> np.ndarray:
    """Return the latest complete stretches of past values, as windows.

    The Schaake shuffle's historical windows: every stretch of `steps`
    consecutive steps of all series, newest first, one step apart, skipping
    any with a value missing.

    Parameters
    ----------
    values : numpy.ndarray
        Past values `[series, time]`, oldest first, NaN where missing.
    steps : int
        Steps a window spans, the forecast's.
    n_windows : int
        Most windows to return.

    Returns
    -------
    numpy.ndarray
        Windows `[n, series, steps]`, `n <= n_windows`.
    """
    values = np.asarray(values, dtype=float)
    out = []
    for end in range(values.shape[1], steps - 1, -1):
        window = values[:, end - steps : end]
        if np.isfinite(window).all():
            out.append(window)
            if len(out) == n_windows:
                break
    return np.asarray(out).reshape(len(out), values.shape[0], steps)

    def energy_score(draws: np.ndarray, observed: np.ndarray) -> float:
    r"""Return the energy score of trajectories on the observed cells.

    $\mathrm{ES} = E\|X - y\| - \tfrac12 E\|X - X'\|$ over the cells
    where `observed` is finite, the second term from each draw and the
    next (Gneiting et al. 2008), so the cost is linear in the draws. It
    estimates the same score as `nowcasteval.scoring.energy_sample`'s fair
    estimator, without its cost quadratic in the draws, and is noisier.

    Parameters
    ----------
    draws : numpy.ndarray
        Trajectories `[n, ...]`.
    observed : numpy.ndarray
        Values, the shape of one trajectory, NaN where not observed.

    Returns
    -------
    float
        The score (lower is better), NaN with no observed cell.
    """
    draws = np.asarray(draws, dtype=float)
    observed = np.asarray(observed, dtype=float)
    ok = np.isfinite(observed)
    if not ok.any():
        return float("nan")
    x = draws[:, ok]
    to_truth = np.linalg.norm(x - observed[ok], axis=1).mean()
    spread = np.linalg.norm(x - np.roll(x, 1, axis=0), axis=1).mean()
    return float(to_truth - 0.5 * spread)

    def choose_coupling(
    quantiles: np.ndarray,
    levels: np.ndarray,
    observed: np.ndarray,
    days: np.ndarray,
    day: int,
    history: np.ndarray | None,
    starts: np.ndarray | None,
    n_samples: int,
    seed: int,
    candidates: tuple[tuple[str, int | None], ...] = CANDIDATES,
) -> dict[str, Any]:
    """Return the coupling, with its fit, whose draws scored best of late.

    Each earlier origin with at least `MIN_EARLIER` before it and a value
    reported since is scored: every candidate is fitted from what came
    before it (the gaussian from the normal scores, `pit_scores`, of the
    earlier origins at most `window` days before it; the Schaake shuffle
    from windows of
    `history` before the origin's first step), that origin's quantiles are
    coupled (`couple`) on the steps up to its last reported one, with the
    same seed for every candidate, and the draws are scored against what
    was reported (`energy_score`). The candidate with the lowest mean
    score over origins every candidate could be fitted at wins; ties go
    to the earlier candidate; a candidate that could not be fitted at one
    of the origins scored drops out. The winner is then fitted from
    everything before the forecast. With nothing to compare on, the
    gaussian over the longest window is used when there are `MIN_EARLIER`
    origins to fit it from, else the Schaake shuffle when the history has
    windows, else the gaussian (independence), and the record says so
    (`chosen_by`).

    Parameters
    ----------
    quantiles : numpy.ndarray
        The model's earlier forecasts' quantiles `[origins, series, steps,
        Q]` at `levels`, oldest first, on the forecast's series and steps.
    levels : numpy.ndarray
        Increasing quantile levels, `[Q]`.
    observed : numpy.ndarray
        What has been reported of their targets, `[origins, series,
        steps]`, NaN where not final.
    days : numpy.ndarray
        Each origin's day, `[origins]`, increasing (any integer count of
        days, such as an ordinal date).
    day : int
        The forecast's day, on the same count.
    history : numpy.ndarray or None
        The series' reported values `[series, time]` before the forecast's
        first step, NaN where missing; None for no Schaake shuffle.
    starts : numpy.ndarray or None
        Each origin's first step as a column of `history`, `[origins]`.
    n_samples : int
        Trajectories, and the most Schaake windows.
    seed : int
        Seed for the scores' and the draws' uniforms.
    candidates : tuple of (str, int or None)
        Each candidate's method and gaussian window in days.

    Returns
    -------
    dict
        The chosen fit, which `couple` takes, with `window`, the mean
        `energy` of each candidate compared, by name, `n_eval` (origins
        scored on),
        `chosen_by` (`energy` or `fallback`) and, for the Schaake shuffle,
        `n_windows`.

    Raises
    ------
    ValueError
        With no candidates, `days` not one per origin, or a history
        without `starts` for each origin.
    """
    if not candidates:
        raise ValueError("choose_coupling needs candidates")
    quantiles = np.asarray(quantiles, dtype=float)
    observed = np.asarray(observed, dtype=float)
    levels = np.asarray(levels, dtype=float)
    n_origins, _, steps = observed.shape
    if history is not None:
        history = np.asarray(history, dtype=float)
        if starts is None or len(starts) != n_origins:
            raise ValueError("a history needs starts, one per origin")
    first = np.zeros(n_origins, dtype=int) if starts is None else starts
    days = np.asarray(days, dtype=int)
    if len(days) != n_origins:
        raise ValueError("days needs one day per origin")
    rng = np.random.default_rng([seed, 1])
    scores = pit_scores(quantiles, levels, observed, rng)
    now = {
        c: _candidate_fit(*c, scores, day - days, history, steps, n_samples)
        for c in candidates
    }
    usable = [c for c in candidates if now[c] is not None]
    got: dict[tuple[str, int | None], list[float]] = {c: [] for c in usable}
    n_eval = 0
    for k in range(MIN_EARLIER, n_origins):
        reported = np.flatnonzero(np.isfinite(observed[k]).any(axis=0))
        if not len(reported):
            continue
        n = int(reported[-1]) + 1
        past = None if history is None else history[:, : int(first[k])]
        n_eval += 1
        for c in usable:
            ages = days[k] - days[:k]
            fit = _candidate_fit(*c, scores[:k], ages, past, steps, n_samples)
            if fit is None:
                got[c].append(np.nan)
                continue
            if "windows" in fit:
                fit["windows"] = fit["windows"][:, :, :n]
            draws = couple(
                quantiles[k][:, :n], levels, c[0], fit, n_samples, seed + k
            )
            got[c].append(energy_score(draws, observed[k][:, :n]))
    # Compared on the same origins: a candidate with no fit at one of them
    # (the Schaake shuffle with too little history then) drops out.
    scored = [c for c in usable if n_eval and np.isfinite(got[c]).all()]
    if scored:
        mean = {c: float(np.mean(got[c])) for c in scored}
        chosen = min(scored, key=lambda c: (mean[c], scored.index(c)))
        chosen_by = "energy"
    else:
        mean, chosen_by = {}, "fallback"
        gaussian = [c for c in usable if c[0] == "gaussian"]
        schaake = [c for c in usable if c[0] == "schaake"]
        longest = max(gaussian, key=lambda c: c[1] or 0, default=None)
        if longest is not None and (n_origins >= MIN_EARLIER or not schaake):
            chosen = longest
        else:
            chosen = (schaake or usable)[0]
    fit = now[chosen] or _candidate_fit(
        *chosen, scores, day - days, history, steps, n_samples
    )
    assert fit is not None
    out = {
        **fit,
        "window": chosen[1],
        "energy": {_label(*c): e for c, e in mean.items()},
        "n_eval": n_eval,
        "chosen_by": chosen_by,
    }
    if "windows" in fit:
        out["n_windows"] = len(fit["windows"])
    return out
