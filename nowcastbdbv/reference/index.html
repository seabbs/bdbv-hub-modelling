    The BDBV outbreak as a case study: the hub's data, zones and delays.

    Import from the modules themselves, for example from nowcastbdbv.data.tidy import load_target.

    nowcastbdbv.data ¶

    The hub data: download, tidy and load its streams.

    • tidy: download the hub's data streams and tidy and load them.
    • delays: empirical onset-to-report delays.
    • zones: the hub's health-zone codes and weekly zone counts.
    • zone_map: the health-zone map.

    nowcastbdbv.data.tidy ¶

    Download, tidy and load the BDBV Modeling Hub data.

    The hub publishes these streams (see DATA_STREAMS):

    • target-data/time-series.csv: weekly confirmed cases by onset week, for the country, provinces and health zones, one block per as_of vintage. This is the target, observed plus imputed onsets.
    • auxiliary-data/daily-confirmed-case.csv: daily confirmed cases by onset day, split into observed and imputed onsets, per vintage.
    • auxiliary-data/weekly-reporting-delays-by-province.csv: one row per case with a known onset, giving its province, onset week and onset-to-report delay. Only the latest vintage is in the file. Earlier vintages are in archived-delay-data/ or only in the git history, so fetch_hub_data walks the history to find them all.
    • auxiliary-data/locations.csv and health-zone-code-mapping.csv: location codes, names and parents.
    • hub-config/: the round configuration, used for validation.

    The hub publishes no deaths, suspected cases, report-date series or populations. Province populations come from a cited public source instead (PROVINCE_POPULATIONS).

    A hub release is labelled by its as_of date. From the 2026-11-01 release on, as_of is the extraction date, one day after the last reported case. The hub states that earlier releases were labelled by the last reported case. release_id maps a label, as published or as the hub may rewrite it, to the label we key releases by, so results, the submission ledger and rounds keep their keys.

    The tidy tables written by build_tidy_data are small and committed under data/. Raw downloads go to data/raw/ and are not committed.

    def release_id(as_of: dt.date | str) -> str:
    """Return the label we key a hub release by.

    A label in `ORIGINAL_AS_OF` is kept. If the hub rewrites one a day
    later, it maps back. Other labels are kept.

    Parameters
    ----------
    as_of : datetime.date or str
        The release's `as_of`, as published or as rewritten.

    Returns
    -------
    str
        The release's label, ISO formatted.

    Examples
    --------
    >>> release_id("2026-10-25"), release_id("2026-11-01")
    ('2026-10-24', '2026-11-01')
    """
    day = as_date(as_of)
    before = (day - dt.timedelta(days=1)).isoformat()
    return before if before in ORIGINAL_AS_OF else day.isoformat()

    def fetch_hub_data(
    raw_dir: Path | str,
    ref: str = "main",
    token: str | None = None,
) -> dict[str, Any]:
    """Download the hub's data, including every vintage of the delay data.

    The latest versions of the files in `DATA_STREAMS` are taken from
    `ref`.
    For the delay data, every commit that touched `auxiliary-data/` is
    searched for delay files, and one file is kept per `as_of` vintage
    (from the most recent commit that has it).

    Parameters
    ----------
    raw_dir : pathlib.Path or str
        Where to save the files. Delay files go to `raw_dir/delays/`.
    ref : str
        Git ref of the hub to take the latest files from.
    token : str, optional
        GitHub token for the API. Defaults to `GITHUB_TOKEN` if set.
        Unauthenticated requests are limited to 60 an hour, which is
        enough for one run.

    Returns
    -------
    dict
        A manifest with the commit SHA used and the source of each delay
        vintage. It is also written to `raw_dir/manifest.json`.
    """
    token = token or os.environ.get("GITHUB_TOKEN")
    raw = Path(raw_dir)
    (raw / "delays").mkdir(parents=True, exist_ok=True)
    head = _api(f"repos/{HUB_REPO}/commits/{ref}", token)["sha"]
    for path in _LATEST_FILES:
        dest = raw / path
        dest.parent.mkdir(parents=True, exist_ok=True)
        dest.write_bytes(_raw(head, path, token))

    commits = _api(
        f"repos/{HUB_REPO}/commits?path=auxiliary-data&sha={ref}&per_page=100",
        token,
    )
    delay_sources: dict[str, dict[str, str]] = {}
    for commit in commits:  # newest first
        sha = commit["sha"]
        tree = _api(f"repos/{HUB_REPO}/git/trees/{sha}?recursive=1", token)
        paths = [
            item["path"]
            for item in tree["tree"]
            if item["path"].startswith("auxiliary-data/")
            and "reporting-delays" in item["path"]
            and item["path"].endswith(".csv")
        ]
        for path in paths:
            content = _raw(sha, path, token)
            as_of = _delay_as_of(content, path)
            if as_of in delay_sources:
                continue
            (raw / "delays" / f"{as_of}.csv").write_bytes(content)
            delay_sources[as_of] = {"commit": sha, "path": path}
    manifest = {
        "repo": HUB_REPO,
        "ref": ref,
        "commit": head,
        "downloaded": dt.datetime.now(dt.UTC).isoformat(timespec="seconds"),
        "delay_vintages": dict(sorted(delay_sources.items())),
    }
    (raw / "manifest.json").write_text(json.dumps(manifest, indent=2) + "\n")
    return manifest

    def tidy_locations(locations_csv: Path | str) -> pd.DataFrame:
    """Tidy the country and province rows of `locations.csv`.

    Adds the 2019 population (`PROVINCE_POPULATIONS`, summed for the
    country) and the province centre where known.

    Parameters
    ----------
    locations_csv : pathlib.Path or str
        Path to the hub's `auxiliary-data/locations.csv`.

    Returns
    -------
    pandas.DataFrame
        Columns `location`, `location_name`, `location_type`,
        `population`, `latitude` and `longitude`.
    """
    loc = pd.read_csv(locations_csv)
    loc = loc.loc[loc["location_type"].isin(["country", "province"])].copy()
    pop = pd.Series(PROVINCE_POPULATIONS, dtype="int64")
    loc["population"] = loc["location"].map(pop)
    loc.loc[loc["location"] == "CD", "population"] = int(pop.sum())
    loc["population"] = loc["population"].astype("int64")
    loc["latitude"] = loc["location"].map(
        {k: v[0] for k, v in PROVINCE_CENTRES.items()}
    )
    loc["longitude"] = loc["location"].map(
        {k: v[1] for k, v in PROVINCE_CENTRES.items()}
    )
    cols = [
        "location",
        "location_name",
        "location_type",
        "population",
        "latitude",
        "longitude",
    ]
    return loc[cols].reset_index(drop=True)

    def tidy_daily(daily_csv: Path | str, level: str = "province") -> pd.DataFrame:
    """Tidy daily onsets by vintage, for one level of geography.

    Parameters
    ----------
    daily_csv : pathlib.Path or str
        Path to the hub's `auxiliary-data/daily-confirmed-case.csv`.
    level : {"province", "zone"}
        The country and provinces, or the health zones, which are kept
        apart for a later zone stage.

    Returns
    -------
    pandas.DataFrame
        Columns `vintage`, `location`, `onset_date`, `observed` and
        `imputed`. Only days with at least one case are present.
    """
    types = {"province": ["country", "province"], "zone": ["health zone"]}
    if level not in types:
        raise ValueError(f"Unknown level {level!r}: use {sorted(types)}.")
    raw = pd.read_csv(daily_csv)
    raw = raw.loc[raw["location_type"].isin(types[level])]
    raw = raw.assign(
        kind=np.where(raw["imputed"].astype(bool), "imputed", "observed")
    )
    wide = raw.pivot_table(
        index=["as_of", "location", "target_end_date"],
        columns="kind",
        values="observation",
        aggfunc="sum",
        fill_value=0,
    ).reset_index()
    for col in ("observed", "imputed"):
        if col not in wide.columns:
            wide[col] = 0
    wide = wide.rename(
        columns={"as_of": "vintage", "target_end_date": "onset_date"}
    )
    wide["vintage"] = wide["vintage"].map(release_id)
    wide.columns.name = None

        (daily["vintage"] == vintage) & (daily["location"] == location)
    ]

        rows.assign(onset_date=pd.to_datetime(rows["onset_date"]))
        .set_index("onset_date")[["observed", "imputed"]]

            "observed",
            "imputed",
            "total",

    """Return how each week's count changed between consecutive releases.

    Parameters
    ----------
    counts : pandas.DataFrame
        Columns `vintage`, `cutoff`, `location`, `target_end_date` and
        `total`. A place and week missing from a release has no cases in
        it, as in the health zone table.

    Returns
    -------
    pandas.DataFrame
        One row per release after the first, place and week of the earlier
        release that has cases in either, with `vintage` (the later
        release), `previous`, `location`, `target_end_date`,
        `weeks_before` (whole weeks from the week to the epi week of the
        earlier release's cut-off), `previous_total`, `total` and `change`.
    """
    keys = ["location", "target_end_date"]
    releases = sorted(counts["vintage"].unique())
    cutoffs = counts.groupby("vintage")["cutoff"].first()
    frames: list[pd.DataFrame] = []
    for prev, vin in pairwise(releases):
        old = counts.loc[counts["vintage"] == prev, [*keys, "total"]]
        new = counts.loc[
            (counts["vintage"] == vin)
            & counts["target_end_date"].between(
                old["target_end_date"].min(), old["target_end_date"].max()
            ),
            [*keys, "total"],
        ]
        pair = old.merge(
            new, on=keys, how="outer", suffixes=("_previous", "")
        ).fillna({"total_previous": 0, "total": 0})
        pair = pair.loc[(pair["total_previous"] > 0) | (pair["total"] > 0)]
        frames.append(
            pair.rename(columns={"total_previous": "previous_total"}).assign(
                vintage=vin, previous=prev
            )
        )
    out = pd.concat(frames, ignore_index=True)
    out[["previous_total", "total"]] = out[["previous_total", "total"]].astype(
        "int64"
    )
    out["change"] = out["total"] - out["previous_total"]
    out["weeks_before"] = _weeks_before(
        out["previous"].map(cutoffs), out["target_end_date"]
    )
    cols = [
        "vintage",
        "previous",
        *keys,
        "weeks_before",
        "previous_total",
        "total",
        "change",
    ]
    return out[cols]

        """Fit to the delay line data of one vintage.

        Parameters
        ----------
        delays : pandas.DataFrame
            Delay line data with `location`, `onset_week` and `delay`.
        vintage : str
            Not used: this estimator ignores right truncation.

        Returns
        -------
        EmpiricalDelayFit
            Pooled and per-cell pmfs.

        weeks = sorted(set(delays["onset_week"].astype(str)))
        half = self.window_weeks // 2
        by_cell: dict[tuple[str, str], NDArray[np.float64]] = {}
        week_idx = {w: i for i, w in enumerate(weeks)}
        idx = delays["onset_week"].astype(str).map(week_idx).to_numpy()
        for i, week in enumerate(weeks):
            in_window = np.abs(idx - i) <= half
            window = delays.loc[in_window]
            pooled_w = empirical_pmf(window["delay"], self.max_delay)
            n_w = int(pooled_w.sum() > 0)
            centre = pooled_w if n_w else pooled
            for loc, rows in window.groupby("location"):

                    (
                        (rows["delay"] >= 0) & (rows["delay"] <= self.max_delay)
                    ).sum()
                )
                by_cell[(str(loc), week)] = (
                    n * counts + self.prior_weight * centre
                ) / (n + self.prior_weight)

    def score_zones(
    target: pd.DataFrame,
    delays: pd.DataFrame | None,
    daily: pd.DataFrame,
    daily_zones: pd.DataFrame,
    run: ZoneForecasts,
    config: BacktestConfig | None = None,
) -> tuple[pd.DataFrame, pd.DataFrame, pd.DataFrame]:
    """Score zone forecasts on later releases.

    Two targets are scored, as for the provinces (`release_backtest`): what
    each later release reported, and the complete weeks of the latest
    release, decided by `complete_weeks` from the zone counts of every
    release, a week with no onset counting as 0.

    Parameters
    ----------
    target : pandas.DataFrame
        The province target table (`load_target`).
    delays : pandas.DataFrame or None
        The delay table (`load_delays`).
    daily : pandas.DataFrame
        The daily table of the country and provinces (`load_daily`).
    daily_zones : pandas.DataFrame
        The daily zone table (`load_daily_zones`).
    run : ZoneForecasts
        The forecasts, such as the walks' stored zone runs. Every
        release is still read for the truth and the complete weeks.
    config : BacktestConfig, optional
        The seed of the reported-count draws. Defaults to
        `BacktestConfig()`.

    Returns
    -------
    tuple of pandas.DataFrame
        The scores, `score_forecasts` output with `model`, `origin`,
        `target` (`"reported"` or `"complete"`), `truth_vintage` and
        `horizon` in weeks
        from the reference date, one row per model, origin, later
        vintage, zone and week.
        Then `energy_scores` of every model's zones by `origin`, `target`
        and `truth_vintage`, from the same drawn samples. Then the zone
        forecasts, as submitted, under each model's name: `model`,
        `origin`, `reference_date`, `location`, `target_end_date`,
        `sample` and `value`.
    """
    config = config or BacktestConfig()
    real = target.loc[target["kind"] == "real"]
    vintages = real_vintages(real)
    latest = vintages[-1]
    done = _complete_zone_weeks(daily_zones, daily, vintages)
    latest_counts = weekly_counts(daily_zones, latest)
    runs = run.runs.astype({"origin": str})
    frames, energy, kept = [], [], []
    for vintage in vintages[:-1]:
        here = runs.loc[runs["origin"] == vintage, "model"].astype(str)
        if here.empty:
            continue
        task = _zone_task(real, delays, daily, config, vintage)
        counts = weekly_counts(daily_zones, vintage)
        forecasts: dict[str, pd.DataFrame] = {}
        summaries: dict[str, pd.DataFrame | None] = {}
        for name in here:
            mine = (run.samples["key"] == name) & (
                run.samples["origin"].astype(str) == vintage
            )
            forecasts[name] = (
                run.samples.loc[mine]
                .drop(columns=["key", "origin"])
                .reset_index(drop=True)
            )
            summaries[name] = model_origin_rows(run.delays, name, vintage)
            kept.append(
                forecasts[name].assign(
                    model=name,
                    origin=vintage,
                    reference_date=task.reference_date.isoformat(),
                )
            )
        if not forecasts:
            continue
        reported = counts.rename(columns={"total": "reported"})[
            ["location", "target_end_date", "reported"]
        ]
        # Each model draws from its own stream, as in release_backtest.
        rngs = {name: task.rng(name) for name in forecasts}
        later = [v for v in vintages if v > vintage]
        by_model = {
            key: reported_samples(
                samples,
                summaries[key],
                key,
                vintage,
                later,
                daily,
                reported,
                rngs[key],
            )
            for key, samples in forecasts.items()
        }
        for o2 in later:
            drawn = [
                draws[o2].assign(
                    model=key,
                    origin=vintage,
                    target="reported",
                    truth_vintage=o2,
                )
                for key, draws in by_model.items()
                if o2 in draws
            ]
            if not drawn:
                continue
            both = pd.concat(drawn, ignore_index=True)
            truth = _zone_truth(weekly_counts(daily_zones, o2), both, o2)
            both["horizon"] = _weeks(both, task)
            frames.append(score_forecasts(both, truth, keys=_KEYS))
            energy.append(energy_scores(both, truth, unit=_KEYS[1:4]))
        # The final-count samples on the latest release's complete weeks.
        final = pd.concat(
            [
                samples.assign(
                    model=key,
                    origin=vintage,
                    target="complete",
                    truth_vintage=latest,
                )
                for key, samples in forecasts.items()
            ],
            ignore_index=True,
        )
        truth = _zone_truth(latest_counts, final, latest).merge(done)
        final["horizon"] = _weeks(final, task)
        frames.append(score_forecasts(final, truth, keys=_KEYS))
        energy.append(energy_scores(final, truth, unit=_KEYS[1:4]))
    if not frames:
        # No origin forecast, as in a block of origins every model skips.
        return pd.DataFrame(), pd.DataFrame(), pd.DataFrame()
    return (
        pd.concat(frames, ignore_index=True),
        pd.concat(energy, ignore_index=True),
        pd.concat(kept, ignore_index=True)[
            [
                "model",
                "origin",
                "reference_date",
                "location",
                "target_end_date",
                "sample",
                "value",
            ]
        ],
    )
